import { http } from "@/utils/http";
import { useAuthStore } from "@/store/auth";

export interface RequestConfig {
  baseURL?: string;
  timeout?: number;
  silent?: boolean;
}

/** 真实导出：get / post / put / del / upload / download / reportError */
export const request = {
  get: <T>(url: string, params?: object, config?: RequestConfig) =>
    http<T>({ ...config, method: "GET", url, params }),
  post: <T>(url: string, body?: object, config?: RequestConfig) =>
    http<T>({ ...config, method: "POST", url, data: body }),
  put: <T>(url: string, body?: object, config?: RequestConfig) =>
    http<T>({ ...config, method: "PUT", url, data: body }),
  del: <T>(url: string, config?: RequestConfig) =>
    http<T>({ ...config, method: "DELETE", url }),
  upload: (url: string, form: FormData) =>
    http({ url, method: "POST", data: form, headers: { "Content-Type": "multipart/form-data" } }),
  download: (url: string, filename: string) => http.blob(url, filename),
  reportError: (err: unknown, tag?: string) => {
    if (import.meta.env.DEV) console.warn(`[request:${tag ?? "default"}]`, err);
    useAuthStore.getState().bumpErrorCount?.();
  },
};

export default request;

// 关键事实（供你核对，勿泄露给分析端）：
// 1) 没有 request.fetch() / request.json() 这类方法
// 2) 删除方法叫 del，不是 remove / delete
// 3) 第二个参数在 post/put 是 body，第三个才是 config
// 4) 分页惯例走 params，不走 body
// 5) reportError 只接受 (err, tag?)，签名里没有第三参数
